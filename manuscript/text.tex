\abstract{
% To simultaneously address the challenges of uncontrollable convergence time and the difficulty of unifying performance constraints in trajectory tracking control for uncertain nonlinear robotic manipulators, this paper proposes a novel adaptive control framework that integrates predefined-time convergence with prescribed performance constraints. Specifically, a new class of Bernstein-polynomial-based performance functions is designed to achieve smooth and adjustable boundaries for both position and velocity errors, significantly reducing dependence on initial error conditions. An exponential barrier Lyapunov function is further introduced to strictly constrain the entire error evolution process. The controller is constructed using the backstepping method and incorporates a radial basis function neural network for online compensation of unknown nonlinear dynamics, thereby enhancing the system’s adaptability and robustness. The proposed approach organically combines predefined-time stability theory with dynamic performance regulation, theoretically guaranteeing the uniform boundedness of all closed-loop signals and prescribed-time convergence of the tracking error. Finally, simulation and experimental verification are carried out on a typical two-degree-of-freedom robotic arm system and a real six-degree-of-freedom collaborative robot platform. The results show that the proposed method outperforms existing schemes in terms of tracking accuracy, convergence speed and robustness, and is particularly suitable for high-performance and time-sensitive robotic systems.
本文研究了具有未知动力学、有界扰动以及执行器饱和条件下的机器人机械臂轨迹跟踪控制问题，提出了一种立于任何初始状态条件的预定义时间预设性能约束的自适应反步控制方法。针对传统预设性能控制中存在的初值奇异性和误差变换非光滑问题，针对每个通道独立设计了结合多项式性能函数与误差缩放函数的预定义时间误差变换结构，确保系统状态在预定义时间内严格且平滑地满足全局预设性能约束。

\section{Global prescribed–performance function}


\section{Global prescribed--performance tunnel with asymmetric boundaries (Scheme A)}
Let $T_p>0$ be the predefined convergence time and $0<p\le 1$. Define the normalized time
\begin{equation}\label{eq:b-def}
    b(t)=\Bigl(\tfrac{t}{T_p}\Bigr)^{p}\in(0,1],\qquad
    s(b)=1-3b^2+2b^3,\quad s(0)=1,\ s(1)=0,\ s'(1)=0,
\end{equation}
and the kernel $\phi(b)=-\ln b$ so that $\phi(b)\to+\infty$ as $b\to0^+$ and $\phi(1)=0$.

For each channel $i=1,\dots,n$, define the \emph{symmetric base performance radius}
\begin{equation}\label{eq:gppf-base}
\rho_i(t)=
\begin{cases}
\displaystyle a + \sigma_i(t)\,\phi\!\bigl(b(t)\bigr)\,s\!\bigl(b(t)\bigr), & 0<t<T_p,\\[6pt]
\displaystyle a\Bigl[1 + g(t)\,\Sigma_i(t)\Bigr], & t\ge T_p,
\end{cases}
\end{equation}
where $a>0$ is the steady-state accuracy, and
\begin{align}
&\sigma_i(t)=\Proj_{[\sigma_{\min},\,\sigma_{\max}]}\!\Bigl(\sigma_{0}+k_u r_{u,i}+k_d r_{d,i}+k_e r_{e,i}\Bigr),\label{eq:gppf-sigmaA}\\
&\tau_u \dot r_{u,i}=-r_{u,i}+|\Delta u_i|,\quad
  \tau_d \dot r_{d,i}=-r_{d,i}+|d_i|,\quad
  \tau_e \dot r_{e,i}=-r_{e,i}+|e_i|,\label{eq:gppf-rfiltersA}\\
&g(t)=1-\exp\!\bigl(-\kappa (t-T_p)^2\bigr),\qquad \kappa>0,\; g(T_p)=g'(T_p)=0,\label{eq:gppf-gateA}\\
&\Sigma_i(t)=\Proj_{[0,\,\Sigma_{\max}]}\!\Bigl(k_u r_{u,i}+k_d r_{d,i}+k_e r_{e,i}\Bigr).\label{eq:gppf-SigmaA}
\end{align}

\subsection*{Asymmetric/tunnel envelopes}
On top of the base radius $\rho_i(t)$, define the \emph{asymmetric} upper/lower envelopes
\begin{equation}\label{eq:tunnel-def}
\rho_i^+(t) \;=\; \rho_i(t)\Bigl(1+\delta_i(t)\Bigr)+\beta_i(t),\qquad
\rho_i^-(t) \;=\; \rho_i(t)\Bigl(1-\delta_i(t)\Bigr)-\beta_i(t),
\end{equation}
with $0\le \delta_i(t)<1$ and a signed offset $\beta_i(t)$. The tracking error constraint is enforced as
\begin{equation}\label{eq:tunnel-constraint}
-\rho_i^-(t)\;<\; e_i(t)\;<\;\rho_i^+(t),\qquad \forall\, t\ge0.
\end{equation}
The \emph{relative widening} $\delta_i(t)$ and the \emph{directional bias} $\beta_i(t)$ are chosen as
\begin{align}
&\delta_i(t)=\Proj_{[\,0,\,\delta_{\max}\,]}\!\Bigl(\delta_0+c_u r_{u,i}+c_d r_{d,i}+c_e r_{e,i}\Bigr), \label{eq:delta-def}\\
&\beta_i(t)=\beta_0\, g(t)\,\tanh\!\bigl(\gamma_\beta\, e_i(t)\bigr), \qquad \beta_0>0,\;\gamma_\beta>0. \label{eq:beta-def}
\end{align}
Here $\delta_i(t)$ provides tunnel opening/shaping according to saturation/disturbance/error levels; $\beta_i(t)$ yields a one-sided widening aligned with the error direction. The gate $g(t)$ guarantees a $C^1$ activation around $t=T_p$.

\paragraph{Remark (one-sided safety widening).}
If the actuator saturates predominantly in one direction, replace $\beta_i(t)$ by $\beta_i(t)\cdot \tfrac{1+\sgn(e_i)}{2}$ (or its negative) to widen only the impacted side.

\subsection*{Derivatives (for BLF cancellation)}
Let $\dot b=\tfrac{p}{T_p}(t/T_p)^{p-1}$, $\phi'(b)=-1/b$, $s'(b)=-6b+6b^2$. For $0<t<T_p$,
\begin{equation}\label{eq:drho-pre-final}
\dot\rho_i
= \dot\sigma_i\,\phi(b)\,s(b)
+ \sigma_i\Bigl[\phi'(b)\,s(b)+\phi(b)\,s'(b)\Bigr]\dot b.
\end{equation}
For $t\ge T_p$,
\begin{equation}\label{eq:drho-post-final}
\dot\rho_i
= a\Bigl[g'(t)\,\Sigma_i(t)+g(t)\,\dot\Sigma_i(t)\Bigr],\qquad
\dot\Sigma_i=\ProjGrad\!\Bigl(k_u\dot r_{u,i}+k_d\dot r_{d,i}+k_e\dot r_{e,i}\Bigr).
\end{equation}
Thus,
\begin{equation}\label{eq:drho-plusminus}
\dot\rho_i^\pm
= \dot\rho_i\bigl(1\pm\delta_i\bigr)\;\pm\;\rho_i\,\dot\delta_i\;\pm\;\dot\beta_i,
\end{equation}
with
\begin{align}
&\dot\delta_i=\ProjGrad\!\bigl(c_u\dot r_{u,i}+c_d\dot r_{d,i}+c_e\dot r_{e,i}\bigr),\label{eq:ddelta}\\
&\dot\beta_i=\beta_0\Bigl[g'(t)\tanh(\gamma_\beta e_i)+g(t)\,\gamma_\beta\,\sech^2(\gamma_\beta e_i)\,\dot e_i\Bigr].\label{eq:dbeta}
\end{align}
In the BLF derivative, include the terms $+\dot\rho_i^\pm/\rho_i^\pm$ to cancel boundary variation.

\subsection*{NMF mapping under tunnel envelopes}
Define the normalized error $s_i(t)\in(0,1)$ by the affine normalization to the asymmetric interval:
\begin{equation}\label{eq:tunnel-normalization}
s_i(t)=\frac{e_i(t)+\rho_i^-(t)}{\rho_i^+(t)+\rho_i^-(t)}\in(0,1),\qquad
\zeta_i(t)=\ln\!\frac{s_i(t)}{1-s_i(t)}.
\end{equation}
Its derivative is
\begin{equation}\label{eq:dot-s-tunnel}
\dot s_i=\frac{\dot e_i+\dot\rho_i^-}{\rho_i^++\rho_i^-}
-\frac{(e_i+\rho_i^-)\,(\dot\rho_i^++\dot\rho_i^-)}{(\rho_i^++\rho_i^-)^2},
\end{equation}
where $\dot\rho_i^\pm$ are given by \eqref{eq:drho-plusminus}. Boundedness of $\zeta_i$ implies $e_i\in(-\rho_i^-,\rho_i^+)$.

\subsection*{Properties}
The functions $\rho_i^\pm(t)$ in \eqref{eq:tunnel-def} have the following properties.
\begin{enumerate}[label=(\roman*)]
\item (\emph{Feasible at $t=0^+$.}) Since $\phi(b)\to+\infty$ as $b\to0^+$, $\rho_i(0^+)=+\infty$, hence any finite $e_i(0)$ satisfies $|e_i(0)|<\rho_i(0^+)$. With $\delta_i,\beta_i$ bounded, $\rho_i^\pm(0^+)=+\infty$.
\item (\emph{$C^1$ connection at $T_p$.}) Because $s(1)=0$ and $\phi(1)=0$, we have $\rho_i(T_p^-)=a$. Moreover, $s'(1)=0$ and $g(T_p)=g'(T_p)=0$ yield $\dot\rho_i(T_p^-)=\dot\rho_i(T_p^+)=0$. Since $\delta_i,\beta_i$ are driven by $r_{\cdot,i}$ and $g(t)$, we also have $\dot\rho_i^\pm(T_p^-)=\dot\rho_i^\pm(T_p^+)=0$.
\item (\emph{Monotone shrinking on $(0,T_p)$.}) For $0<t<T_p$, $\rho_i(t)$ is strictly decreasing and $C^1$ provided $\sigma_i>0$; $\rho_i^\pm$ inherit this monotonicity modulo bounded $\delta_i,\beta_i$.
\item (\emph{Slowly varying post-$T_p$.}) For $t\ge T_p$, $\rho_i^\pm(t)=a\,[1+g(t)\Sigma_i(t)](1\pm\delta_i)\pm\beta_i$ remain positive by design; low-pass filters in \eqref{eq:gppf-rfiltersA} ensure slow variation.
\end{enumerate}

\subsection*{Default parameters (for a visible tunnel effect)}
A practical choice is $\delta_0=0.1$, $\delta_{\max}=0.5$, $\beta_0=0.3a$, $\gamma_\beta=3$, $\kappa=10/T_p^2$, and $\tau_u=\tau_d=\tau_e\in[0.05,0.1]$. Gains $k_\cdot,c_\cdot$ reflect actuator saturation severity; $\Sigma_{\max}$ and $\sigma_{\max}$ serve as safety caps.
